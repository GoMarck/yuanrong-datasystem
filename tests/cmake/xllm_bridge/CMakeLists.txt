cmake_minimum_required(VERSION 3.20)
project(StaticSdkBridgeValidation LANGUAGES CXX)

foreach(name protobuf::libprotobuf gRPC::grpc++ gflags::gflags brpc::brpc)
    add_library(${name} INTERFACE IMPORTED)
    set_property(TARGET ${name} PROPERTY INTERFACE_LINK_LIBRARIES poison_host_dependency)
endforeach()
if (NOT HOST_ABI MATCHES "^[01]$" OR NOT SDK_ABI MATCHES "^[01]$" OR HOST_ABI STREQUAL SDK_ABI)
    message(FATAL_ERROR "Pass distinct HOST_ABI and SDK_ABI values (0 or 1)")
endif()
add_compile_options("-D_GLIBCXX_USE_CXX11_ABI=${HOST_ABI}")
if (DS_SDK_SOURCE_DIR)
    set(DS_GLIBCXX_USE_CXX11_ABI "${SDK_ABI}")
    add_subdirectory("${DS_SDK_SOURCE_DIR}/cmake/embed" ds-sdk)
else()
    find_package(DatasystemPrivate CONFIG REQUIRED)
endif()

add_library(host_abi_check OBJECT abi.cpp)
target_compile_definitions(host_abi_check PRIVATE EXPECTED_ABI=${HOST_ABI})
add_library(bridge SHARED "${XLLM_SOURCE_DIR}/xllm/core/kv_cache/transfer/yuanrong_store_bridge.cpp" abi.cpp)
add_dependencies(bridge host_abi_check)
target_compile_definitions(bridge PRIVATE EXPECTED_ABI=${SDK_ABI})
target_include_directories(bridge PRIVATE "${XLLM_SOURCE_DIR}/xllm")
target_link_libraries(bridge PRIVATE Datasystem::client_static)
set_target_properties(bridge PROPERTIES OUTPUT_NAME xllm_yuanrong_store_bridge
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
set(exports "${CMAKE_CURRENT_BINARY_DIR}/exports.map")
file(CONFIGURE OUTPUT "${exports}" CONTENT "{ global: xllm_yr_store_*; local: *; };\n")
target_link_options(bridge PRIVATE "LINKER:--as-needed" "LINKER:--exclude-libs,ALL"
    "LINKER:-z,defs" "LINKER:--version-script=${exports}")
set(check "${XLLM_SOURCE_DIR}/cmake/check_yuanrong_bridge.cmake")
set_property(TARGET bridge APPEND PROPERTY LINK_DEPENDS "${exports}" "${check}")
if (NOT AUDIT_CMAKE)
    set(AUDIT_CMAKE "${CMAKE_COMMAND}")
endif()
add_custom_command(TARGET bridge POST_BUILD COMMAND "${AUDIT_CMAKE}" "-DBRIDGE_FILE=$<TARGET_FILE:bridge>"
    "-DREADELF=${CMAKE_READELF}" "-DNM=${CMAKE_NM}" -P "${check}" VERBATIM)
