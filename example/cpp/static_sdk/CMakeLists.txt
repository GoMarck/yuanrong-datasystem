cmake_minimum_required(VERSION 3.20)
project(DatasystemConsumer LANGUAGES CXX)
if (DS_STATIC_SDK_SOURCE_DIR)
    add_subdirectory("${DS_STATIC_SDK_SOURCE_DIR}/cmake/static-sdk" ds-sdk)
else()
    find_package(Datasystem CONFIG REQUIRED)
endif()
add_library(ds_static_sdk_consumer SHARED kv_smoke.cpp)
target_link_libraries(ds_static_sdk_consumer PRIVATE Datasystem::client_static)
if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(ds_static_sdk_consumer PRIVATE -Wl,--no-undefined
        "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/ds_static_sdk_consumer.map")
endif()
add_executable(ds_static_sdk_smoke main.cpp)
target_link_libraries(ds_static_sdk_smoke PRIVATE ds_static_sdk_consumer)
