cmake_minimum_required(VERSION 3.20)
project(DatasystemSourceConsumer LANGUAGES C CXX)

set(DS_SOURCE_PATH "${CMAKE_CURRENT_LIST_DIR}/../../.." CACHE PATH "Datasystem source checkout")
set(DS_HOST_DEPENDENCIES "" CACHE FILEPATH "Host dependency targets, versions and brpc contract")
if (NOT EXISTS "${DS_HOST_DEPENDENCIES}")
    message(FATAL_ERROR "Set DS_HOST_DEPENDENCIES to your host dependency setup; see host-dependencies.cmake.example")
endif()
include("${DS_HOST_DEPENDENCIES}")
set(DS_BUILD_CLIENT_ONLY ON CACHE BOOL "" FORCE)
set(DS_USE_EXTERNAL_DEPS ON CACHE BOOL "" FORCE)
add_subdirectory("${DS_SOURCE_PATH}" datasystem)
add_executable(ds_kv_example main.cpp)
target_link_libraries(ds_kv_example PRIVATE Datasystem::client)
if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(ds_kv_example PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/ds_kv_example.map")
endif()
